import { NextResponse } from "next/server";
import { createHmac } from "node:crypto";
import { prisma } from "@/lib/prisma";

export const runtime = "nodejs";

function sign(payload: string) {
  const secret = process.env.RIDER_SESSION_SECRET;
  if (!secret || secret.length < 32) {
    throw new Error("RIDER_SESSION_SECRET must be at least 32 characters");
  }

  return createHmac("sha256", secret)
    .update(payload)
    .digest("base64url");
}

export async function POST(req: Request) {
  try {
    const body = await req.json();
    const mobile = String(body.mobile ?? "").trim();
    const password = String(body.password ?? "");

    if (!mobile || !password) {
      return NextResponse.json(
        { success: false, error: "Mobile and password required" },
        { status: 400 }
      );
    }

    const rider = await prisma.deliveryBoy.findUnique({
      where: { mobile },
    });

    if (!rider || !rider.password || rider.password !== password) {
      return NextResponse.json(
        { success: false, error: "Invalid mobile or password" },
        { status: 401 }
      );
    }

    if (rider.approvalStatus !== "Approved") {
      return NextResponse.json(
        {
          success: false,
          error: "Your rider account is not approved yet",
          approvalStatus: rider.approvalStatus,
        },
        { status: 403 }
      );
    }

    const payload = Buffer.from(
      JSON.stringify({
        riderId: rider.id,
        exp: Date.now() + 7 * 24 * 60 * 60 * 1000,
      })
    ).toString("base64url");

    const token = `${payload}.${sign(payload)}`;

    const response = NextResponse.json({
      success: true,
      rider: {
        id: rider.id,
        name: rider.name,
        mobile: rider.mobile,
        vehicle: rider.vehicle,
        status: rider.status,
        approvalStatus: rider.approvalStatus,
      },
    });

    response.cookies.set("f2c_rider_session", token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      path: "/",
      maxAge: 7 * 24 * 60 * 60,
    });

    return response;
  } catch (error) {
    console.error("RIDER LOGIN ERROR:", error);

    return NextResponse.json(
      { success: false, error: "Login failed. Check server configuration." },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { createHmac, timingSafeEqual } from "node:crypto";

export const runtime = "nodejs";

function validSignature(payload: string, signature: string) {
  const secret = process.env.RIDER_SESSION_SECRET;
  if (!secret || secret.length < 32) return false;

  const expected = createHmac("sha256", secret)
    .update(payload)
    .digest();

  let supplied: Buffer;
  try {
    supplied = Buffer.from(signature, "base64url");
  } catch {
    return false;
  }

  return supplied.length === expected.length &&
    timingSafeEqual(supplied, expected);
}

export async function GET(req: NextRequest) {
  const token = req.cookies.get("f2c_rider_session")?.value;

  if (!token) {
    return NextResponse.json({ success: false, session: false }, { status: 401 });
  }

  try {
    const parts = token.split(".");
    if (parts.length !== 2 || !validSignature(parts[0], parts[1])) {
      throw new Error("Invalid session");
    }

    const payload = JSON.parse(
      Buffer.from(parts[0], "base64url").toString("utf8")
    );

    if (
      !Number.isInteger(payload.riderId) ||
      !Number.isFinite(payload.exp) ||
      payload.exp <= Date.now()
    ) {
      throw new Error("Expired session");
    }

    return NextResponse.json({
      success: true,
      session: true,
      riderId: payload.riderId,
    });
  } catch {
    const response = NextResponse.json(
      { success: false, session: false },
      { status: 401 }
    );

    response.cookies.set("f2c_rider_session", "", {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      path: "/",
      maxAge: 0,
    });

    return response;
  }
}

import { NextRequest, NextResponse } from "next/server";
import { createHmac, timingSafeEqual } from "node:crypto";
import { prisma } from "@/lib/prisma";

export const runtime = "nodejs";

function getRiderId(req: NextRequest): number | null {
  const token = req.cookies.get("f2c_rider_session")?.value;
  const secret = process.env.RIDER_SESSION_SECRET;

  if (!token || !secret || secret.length < 32) return null;

  const parts = token.split(".");
  if (parts.length !== 2) return null;

  try {
    const expected = createHmac("sha256", secret)
      .update(parts[0])
      .digest();
    const supplied = Buffer.from(parts[1], "base64url");

    if (
      supplied.length !== expected.length ||
      !timingSafeEqual(supplied, expected)
    ) return null;

    const payload = JSON.parse(
      Buffer.from(parts[0], "base64url").toString("utf8")
    );

    if (
      !Number.isInteger(payload.riderId) ||
      !Number.isFinite(payload.exp) ||
      payload.exp <= Date.now()
    ) return null;

    return payload.riderId;
  } catch {
    return null;
  }
}

export async function GET(req: NextRequest) {
  try {
    const riderId = getRiderId(req);

    if (!riderId) {
      return NextResponse.json(
        { success: false, error: "Please log in again", tasks: [] },
        { status: 401 }
      );
    }

    const rider = await prisma.deliveryBoy.findUnique({
      where: { id: riderId },
      select: { id: true, approvalStatus: true },
    });

    if (!rider || rider.approvalStatus !== "Approved") {
      return NextResponse.json(
        { success: false, error: "Rider account is not approved", tasks: [] },
        { status: 403 }
      );
    }

    const tasks = await prisma.deliveryAssignment.findMany({
      where: { deliveryBoyId: riderId },
      include: {
        order: {
          select: {
            id: true,
            customer: true,
            phone: true,
            address: true,
            amount: true,
            status: true,
          },
        },
      },
      orderBy: { assignedAt: "desc" },
    });

    return NextResponse.json({ success: true, tasks });
  } catch (error) {
    console.error("RIDER MY TASKS ERROR:", error);
    return NextResponse.json(
      { success: false, error: "Could not load tasks", tasks: [] },
      { status: 500 }
    );
  }
}

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function GET(req: Request) {
  try {
    const { searchParams } =
      new URL(req.url);

    const riderId = Number(
      searchParams.get("riderId")
    );

    if (!riderId) {
      return NextResponse.json(
        {
          success: false,
          error: "Rider ID required",
        },
        {
          status: 400,
        }
      );
    }

    const assignments =
      await prisma.deliveryAssignment.findMany({
        where: {
          deliveryBoyId: riderId,
        },
        orderBy: {
          assignedAt: "desc",
        },
      });

    const orderIds =
      assignments.map(
        (a) => a.orderId
      );

    const orders =
      orderIds.length > 0
        ? await prisma.order.findMany({
            where: {
              id: {
                in: orderIds,
              },
            },
          })
        : [];

    const orderMap = new Map(
      orders.map((o) => [o.id, o])
    );

    const result =
      assignments.map((a) => ({
        assignmentId: a.id,
        orderId: a.orderId,
        otp: a.otp,
        status: a.status,
        earnings: a.earnings,
        assignedAt: a.assignedAt,

        customer:
          orderMap.get(a.orderId)
            ?.customer || "",

        phone:
          orderMap.get(a.orderId)
            ?.phone || "",

        address:
          orderMap.get(a.orderId)
            ?.address || "",

        amount:
          orderMap.get(a.orderId)
            ?.amount || 0,
      }));

    return NextResponse.json({
      success: true,
      orders: result,
    });
  } catch (error: any) {
    console.error(
      "RIDER ORDERS ERROR",
      error
    );

    return NextResponse.json(
      {
        success: false,
        error: error?.message ?? "Internal Server Error",
      },
      {
        status: 500,
      }
    );
  }
}

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(req: Request) {
  try {
    const body = await req.json();
    const assignmentId = Number(body?.assignmentId);

    if (!assignmentId) {
      return NextResponse.json(
        { success: false, error: "Assignment ID Required" },
        { status: 400 }
      );
    }

    const assignment = await prisma.deliveryAssignment.findUnique({
      where: { id: assignmentId },
    });

    if (!assignment) {
      return NextResponse.json(
        { success: false, error: "Assignment Not Found" },
        { status: 404 }
      );
    }

    if (assignment.status !== "Accepted") {
      return NextResponse.json(
        {
          success: false,
          error: "Order must be Accepted before Pickup",
        },
        { status: 400 }
      );
    }

    const updated = await prisma.deliveryAssignment.update({
      where: { id: assignmentId },
      data: {
        status: "Picked Up",
        pickedAt: new Date(),
      },
    });

    await prisma.order.update({
      where: { id: assignment.orderId },
      data: { status: "Picked Up" },
    });

    return NextResponse.json({
      success: true,
      assignment: updated,
      message: "Order Picked Up Successfully",
    });
  } catch (error: any) {
    console.error("RIDER PICKUP ERROR:", error);

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(request: Request) {
  try {
    const body = await request.json();

    const assignmentId = Number(body?.assignmentId);
    const otp = String(body?.otp || "").trim();

    if (!assignmentId || !otp) {
      return NextResponse.json(
        {
          success: false,
          verified: false,
          error: "Assignment ID and OTP Required",
        },
        { status: 400 }
      );
    }

    const assignment = await prisma.deliveryAssignment.findUnique({
      where: { id: assignmentId },
    });

    if (!assignment) {
      return NextResponse.json(
        { success: false, verified: false, error: "Assignment Not Found" },
        { status: 404 }
      );
    }

    if (assignment.status !== "Otp Sent") {
      return NextResponse.json(
        {
          success: false,
          verified: false,
          error: "Please send delivery OTP first",
        },
        { status: 400 }
      );
    }

    const verified = assignment.otp === otp;

    if (!verified) {
      return NextResponse.json(
        {
          success: false,
          verified: false,
          error: "Invalid OTP",
        },
        { status: 400 }
      );
    }

    return NextResponse.json({
      success: true,
      verified: true,
      assignmentId,
      message: "OTP Verified Successfully",
    });
  } catch (error: any) {
    console.error("RIDER VERIFY OTP ERROR:", error);

    return NextResponse.json(
      {
        success: false,
        verified: false,
        error: error?.message || "OTP Verification Failed",
      },
      { status: 500 }
    );
  }
}
