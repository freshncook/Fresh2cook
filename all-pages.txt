"use client";

import { useEffect, useState } from "react";

export default function ProductsPage() {
  const [products, setProducts] = useState<any[]>([]);
  const [cartCount, setCartCount] = useState(0);
  const [qty, setQty] = useState<Record<number, number>>({});

  useEffect(() => {
    loadProducts();
    loadCartCount();
  }, []);

  async function loadProducts() {
    try {
      const res = await fetch("/api/products");

      if (!res.ok) {
        throw new Error("Failed To Load Products");
      }

      const data = await res.json();

      setProducts(data);

      const initialQty: Record<number, number> = {};

      data.forEach((item: any) => {
        initialQty[item.id] = 1;
      });

      setQty(initialQty);
    } catch (error) {
      console.log(error);
    }
  }

  function loadCartCount() {
    const cart = JSON.parse(
      localStorage.getItem("freshcut-cart") || "[]"
    );

    const totalItems = cart.reduce(
      (sum: number, item: any) =>
        sum + Number(item.qty || 1),
      0
    );

    setCartCount(totalItems);
  }

  function increase(id: number) {
    setQty((prev) => ({
      ...prev,
      [id]: (prev[id] || 1) + 1,
    }));
  }

  function decrease(id: number) {
    setQty((prev) => ({
      ...prev,
      [id]:
        prev[id] > 1
          ? prev[id] - 1
          : 1,
    }));
  }

  function addToCart(product: any) {
    const cart = JSON.parse(
      localStorage.getItem("freshcut-cart") || "[]"
    );

    const index = cart.findIndex(
      (item: any) =>
        item.id === product.id
    );

    if (index >= 0) {
      cart[index].qty =
        Number(cart[index].qty || 1) +
        Number(qty[product.id] || 1);
    } else {
      cart.push({
        ...product,
        qty:
          Number(qty[product.id] || 1),
      });
    }

    localStorage.setItem(
      "freshcut-cart",
      JSON.stringify(cart)
    );

    loadCartCount();

    alert(
      product.name +
        " Added To Cart"
    );
  }

  return (
    <div className="p-6 pb-28">

      <h1 className="text-4xl font-bold mb-6">
        Fresh Cut Products
      </h1>

      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">

        {products.map((product) => (

          <div
            key={product.id}
            className="bg-white rounded-2xl border shadow overflow-hidden"
          >

            <img
              src={
                product.image ||
                "https://images.unsplash.com/photo-1542838132-92c53300491e"
              }
              alt={product.name}
              className="w-full h-40 object-cover"
            />

            <div className="p-4">

              <h3 className="font-bold">
                {product.name}
              </h3>

              <p className="text-sm text-gray-500">
                {product.pack}
              </p>

              <p className="text-green-700 text-xl font-bold mt-2">
                ₹{product.price}
              </p>

              <div className="flex items-center justify-center gap-3 mt-4">

                <button
                  onClick={() =>
                    decrease(product.id)
                  }
                  className="bg-red-500 text-white w-8 h-8 rounded-full"
                >
                  -
                </button>

                <span className="font-bold">
                  {qty[product.id] || 1}
                </span>

                <button
                  onClick={() =>
                    increase(product.id)
                  }
                  className="bg-green-600 text-white w-8 h-8 rounded-full"
                >
                  +
                </button>

              </div>

              <button
                onClick={() =>
                  addToCart(product)
                }
                className="w-full mt-4 bg-green-600 text-white py-2 rounded-xl font-bold"
              >
                Add To Cart
              </button>

            </div>

          </div>

        ))}

      </div>

      <div className="fixed bottom-0 left-0 right-0 bg-green-600 text-white p-4 shadow-xl">

        <a
          href="/cart"
          className="block text-center text-lg font-bold"
        >
          🛒 View Cart ({cartCount})
        </a>

      </div>

    </div>
  );
}

"use client";

import { useEffect, useState } from "react";

export default function CartPage() {
  const [items, setItems] = useState<any[]>([]);

  useEffect(() => {
    loadCart();
  }, []);

  function loadCart() {
    const cart = JSON.parse(
      localStorage.getItem("freshcut-cart") || "[]"
    );

    setItems(cart);
  }

  function updateCart(cart: any[]) {
    localStorage.setItem(
      "freshcut-cart",
      JSON.stringify(cart)
    );

    setItems([...cart]);
  }

  function increase(index: number) {
    const cart = [...items];

    cart[index].qty =
      Number(cart[index].qty || 1) + 1;

    updateCart(cart);
  }

  function decrease(index: number) {
    const cart = [...items];

    if (cart[index].qty > 1) {
      cart[index].qty =
        Number(cart[index].qty) - 1;

      updateCart(cart);
    }
  }

  function removeItem(index: number) {
    const cart = [...items];

    cart.splice(index, 1);

    updateCart(cart);
  }

  const totalItems = items.reduce(
    (sum, item) =>
      sum + Number(item.qty || 1),
    0
  );

  const grandTotal = items.reduce(
    (sum, item) =>
      sum +
      Number(item.price || 0) *
        Number(item.qty || 1),
    0
  );

  function proceed() {
    localStorage.setItem(
      "freshcut-total",
      String(grandTotal)
    );

    localStorage.setItem(
      "freshcut-items",
      String(totalItems)
    );

    window.location.href =
      "/customer-details";
  }

  return (
    <div className="max-w-5xl mx-auto p-6 pb-32">

      <h1 className="text-4xl font-bold mb-6">
        My Cart
      </h1>

      {items.length === 0 && (
        <div className="bg-white rounded-2xl p-8 text-center shadow">
          Cart Empty
        </div>
      )}

      {items.map((item, index) => (

        <div
          key={index}
          className="bg-white rounded-2xl shadow border p-4 mb-4"
        >

          <div className="flex justify-between">

            <div>

              <h2 className="font-bold text-xl">
                {item.name}
              </h2>

              <p className="text-gray-500">
                ₹{item.price}
              </p>

            </div>

            <button
              onClick={() =>
                removeItem(index)
              }
              className="text-red-600 font-bold"
            >
              Remove
            </button>

          </div>

          <div className="flex items-center gap-3 mt-4">

            <button
              onClick={() =>
                decrease(index)
              }
              className="bg-red-500 text-white w-8 h-8 rounded-full"
            >
              -
            </button>

            <span className="font-bold text-lg">
              {item.qty}
            </span>

            <button
              onClick={() =>
                increase(index)
              }
              className="bg-green-600 text-white w-8 h-8 rounded-full"
            >
              +
            </button>

          </div>

          <div className="mt-4 font-bold text-green-700">
            Total: ₹
            {Number(item.price) *
              Number(item.qty)}
          </div>

        </div>

      ))}

      {items.length > 0 && (

        <div className="fixed bottom-0 left-0 right-0 bg-white border-t shadow-2xl p-4">

          <div className="max-w-5xl mx-auto flex items-center justify-between">

            <div>

              <div className="font-bold">
                Items: {totalItems}
              </div>

              <div className="text-2xl font-bold text-green-700">
                ₹{grandTotal}
              </div>

            </div>

            <button
              onClick={proceed}
              className="bg-green-600 text-white px-6 py-3 rounded-xl font-bold"
            >
              Continue
            </button>

          </div>

        </div>

      )}

    </div>
  );
}
