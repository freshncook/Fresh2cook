"use client";

import { useEffect, useState } from "react";

type Product = {
  id: number;
  name: string;
  category: string;
  pack: string;
  mrp: number;
  price: number;
  stock: number;
  image?: string | null;
  description?: string | null;
  active?: boolean;
  bestSeller?: boolean;
};

type ProductForm = {
  name: string;
  category: string;
  pack: string;
  mrp: string;
  price: string;
  stock: string;
  image: string;
  description: string;
  bestSeller: boolean;
  active: boolean;
};

const emptyForm: ProductForm = {
  name: "",
  category: "Vegetables",
  pack: "",
  mrp: "",
  price: "",
  stock: "",
  image: "",
  description: "",
  bestSeller: false,
  active: true,
};

async function getResponseData(response: Response) {
  const text = await response.text();

  if (!text) {
    return {};
  }

  try {
    return JSON.parse(text);
  } catch {
    return {
      error: text,
    };
  }
}
const autoAIDescription = (name: string) => {
  return `Fresh2Cook ${name} - Freshly packed, hygienically processed and ready for your kitchen.`;
};

export default function ProductsManagePage() {
  const [products, setProducts] = useState<Product[]>([]);
  const [loading, setLoading] = useState(true);
  const [adding, setAdding] = useState(false);
  const [saving, setSaving] = useState(false);
  const [errorMessage, setErrorMessage] = useState("");

  const [newProduct, setNewProduct] =
    useState<ProductForm>({
      ...emptyForm,
    });

  const [editingProduct, setEditingProduct] =
    useState<Product | null>(null);

  const [editForm, setEditForm] =
    useState<ProductForm>({
      ...emptyForm,
    });

  useEffect(() => {
    loadProducts();
  }, []);

  async function loadProducts() {
    try {
      setLoading(true);
      setErrorMessage("");

      const response = await fetch(
        "/api/products",
        {
          cache: "no-store",
        }
      );

      const data =
        await getResponseData(response);

      if (!response.ok) {
        throw new Error(
          data?.error ||
            "Failed to load products"
        );
      }

      setProducts(
        Array.isArray(data)
          ? data
          : []
      );
    } catch (error) {
      console.error(error);

      setErrorMessage(
        error instanceof Error
          ? error.message
          : "Failed to load products"
      );
    } finally {
      setLoading(false);
    }
  }

  function openEdit(product: Product) {
    setEditingProduct(product);

    setEditForm({
      name: product.name || "",
      category:
        product.category ||
        "Vegetables",
      pack: product.pack || "",
      mrp: String(
        product.mrp ?? ""
      ),
      price: String(
        product.price ?? ""
      ),
      stock: String(
        product.stock ?? ""
      ),
      image: product.image || "",
      description:
        product.description || "",
      bestSeller:
        product.bestSeller === true,
      active:
        product.active !== false,
    });

    setErrorMessage("");

    window.scrollTo({
      top: 0,
      behavior: "smooth",
    });
  }

  function closeEdit() {
    setEditingProduct(null);

    setEditForm({
      ...emptyForm,
    });
  }

  async function addProduct() {
    setErrorMessage("");

    const name =
      newProduct.name.trim();

    const pack =
      newProduct.pack.trim();

    const mrp = Math.round(
      Number(newProduct.mrp)
    );

    const price = Math.round(
      Number(newProduct.price)
    );

    const stock = Math.round(
      Number(
        newProduct.stock || 0
      )
    );

    if (!name) {
      setErrorMessage(
        "Please enter Product Name."
      );
      return;
    }

    if (!pack) {
      setErrorMessage(
        "Please enter Pack Size."
      );
      return;
    }

    if (
      !Number.isFinite(mrp) ||
      mrp < 0
    ) {
      setErrorMessage(
        "Please enter a valid MRP."
      );
      return;
    }

    if (
      !Number.isFinite(price) ||
      price < 0
    ) {
      setErrorMessage(
        "Please enter a valid Our Price."
      );
      return;
    }

    if (mrp < price) {
      setErrorMessage(
        "MRP cannot be less than Our Price."
      );
      return;
    }

    if (
      !Number.isFinite(stock) ||
      stock < 0
    ) {
      setErrorMessage(
        "Please enter a valid Stock."
      );
      return;
    }

    try {
      setAdding(true);

      const response =
        await fetch(
          "/api/products",
          {
            method: "POST",
            headers: {
              "Content-Type":
                "application/json",
            },
            body: JSON.stringify({
              name,
              category:
                newProduct.category,
              pack,
              mrp,
              price,
              stock,
              image:
                newProduct.image.trim(),
              description:
                newProduct.description.trim(),
              bestSeller:
                newProduct.bestSeller,
              active:
                newProduct.active,
            }),
          }
        );

      const data =
        await getResponseData(
          response
        );

      if (!response.ok) {
        throw new Error(
          data?.error ||
            "Failed to add product"
        );
      }

      setNewProduct({
        ...emptyForm,
      });

      await loadProducts();

      alert(
        "Product Added Successfully"
      );
    } catch (error) {
      console.error(error);

      setErrorMessage(
        error instanceof Error
          ? error.message
          : "Failed to add product"
      );
    } finally {
      setAdding(false);
    }
  }

  async function saveEdit() {
    if (!editingProduct) {
      return;
    }

    setErrorMessage("");

    const name =
      editForm.name.trim();

    const pack =
      editForm.pack.trim();

    const mrp = Math.round(
      Number(editForm.mrp)
    );

    const price = Math.round(
      Number(editForm.price)
    );

    const stock = Math.round(
      Number(
        editForm.stock || 0
      )

      setErrorMessage(
        "Please enter Product Name."
      );
      return;

      return;

    if (
      !Number.isFinite(stock) ||
      stock < 0
    ) {
      setErrorMessage(
        "Please enter a valid Stock."
      );
      return;
    }

    try {
      setSaving(true);

      const response =
        await fetch(
          `/api/products/${editingProduct.id}`,
          {
            method: "PATCH",
            headers: {
              "Content-Type":
                "application/json",
            },
            body: JSON.stringify({
              name,
              category:
                editForm.category,
              pack,
              mrp,
              price,
              stock,
              image:
                editForm.image.trim(),
              description:
                editForm.description.trim(),
              bestSeller:
                editForm.bestSeller,
              active:
                editForm.active,
            }),
          }
        );

      const data =
        await getResponseData(
          response
        );

      if (!response.ok) {
        throw new Error(
          data?.error ||
            "Failed to update product"
        );
      }

      setProducts((current) =>
        current.map(
          (product) =>
            product.id ===
            editingProduct.id
              ? data
              : product
        )
      );

      closeEdit();

      alert(
        "Product Updated Successfully"
      );
    } catch (error) {
      console.error(error);

      setErrorMessage(
        error instanceof Error
          ? error.message
          : "Failed to update product"
      );
    } finally {
      setSaving(false);
    }
  }

  async function deleteProduct(
    product: Product
  ) {
    const confirmed =
      window.confirm(
        `Delete "${product.name}"?`
      );

    if (!confirmed) {
      return;
    }

    try {
      setErrorMessage("");

      const response =
        await fetch(
          `/api/products/${product.id}`,
          {
            method: "DELETE",
          }
        );

      const data =
        await getResponseData(
          response
        );

      if (!response.ok) {
        throw new Error(
          data?.error ||
            "Failed to delete product"
        );
      }

      setProducts((current) =>
        current.filter(
          (item) =>
            item.id !== product.id
        )
      );
    } catch (error) {
      console.error(error);

      setErrorMessage(
        error instanceof Error
          ? error.message
          : "Failed to delete product"
      );
    }
  }

        ...newProduct,
        image:

      });
    };

    reader.readAsDataURL(file);
  }

          </div>
        ) : (
          <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">

            {products.map(
              (product) => {
                const productSaving =
                  Math.max(
                    Number(
                      product.mrp || 0
                    ) -
                      Number(
                        product.price || 0
                      ),
                    0
                  );

                return (
                  <div
                    key={product.id}
                    className={`bg-gradient-to-br from-white to-green-50 rounded-2xl shadow-sm overflow-hidden border hover:shadow-lg transition-all ${
                      product.bestSeller
                        ? "border-yellow-400"
                        : "border-transparent"
                    }`}
                  >

                    <div className="relative">

                      <img
                        src={
                          product.image?.trim()
                            ? product.image
                            : "/placeholder-product.png"
                        }
                        alt={
                          product.name
                        }
                        className="w-full h-24 object-contain bg-gradient-to-br from-white to-green-50 p-1 transition-transform duration-300 hover:scale-110"
                        onError={(e) => {
                          e.currentTarget.src =
                            "/placeholder-product.png";
                        }}
                      />

                      {product.bestSeller && (
                        <div className="absolute top-3 left-3 bg-yellow-400 text-black px-3 py-1 rounded-full text-xs font-bold">
                          BEST SELLER
                        </div>
                      )}

                    </div>

                    <div className="p-3">

                      <h3 className="font-bold text-lg">
                        {product.name}
                      </h3>

                      <p className="text-sm text-gray-500">
  Pack: {product.pack}
</p>

                      <div className="mt-2 rounded-lg bg-gray-50 p-2">

  <div className="flex justify-between text-xs">
    <span>MRP</span>
    <span className="line-through">
      ?{product.mrp || 0}
    </span>
  </div>

  <div className="flex justify-between text-xs mt-1">
    <span>Our Price</span>
    <span className="text-green-600 font-bold">
      ?{product.price}
    </span>
  </div>

  <div className="flex justify-between text-xs mt-1">
    <span>Saving</span>
    <span className="text-red-600 font-bold">
      ?{productSaving}
    </span>
  </div>

</div>  

                      <p className="text-sm text-gray-500 mt-2">
                        Stock:{" "}
                        {product.stock}
                      </p>

  <button
    onClick={() =>
      openEdit(product)
    }
    className="bg-blue-600 text-white py-1 rounded text-[11px] font-semibold"
  >
    Edit
  </button>

  <button
    onClick={() =>
      toggleBestSeller(product)
    }
    className={`py-1 rounded text-[11px] font-semibold ${
      product.bestSeller
        ? "bg-yellow-400 text-black"
        : "bg-green-600 text-white"
    }`}
  >
    Best
  </button>

  <button
    onClick={() =>
      toggleActive(product)
    }
    className="bg-gray-700 text-white py-1 rounded text-[11px]"
  >
    {product.active === false
      ? "On"
      : "Off"}
  </button>

  <button
    onClick={() =>
      deleteProduct(product)
    }
    className="bg-red-600 text-white py-1 rounded text-[11px]"
  >
    Del
  </button>

</div>


                  </div>
                );
              }
            )}

          </div>
        )}

      </div>
    </div>
  );
}
