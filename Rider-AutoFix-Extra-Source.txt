import { NextRequest, NextResponse } from "next/server";
import { createHmac, timingSafeEqual } from "node:crypto";
import { prisma } from "@/lib/prisma";

export const runtime = "nodejs";

function getAuthenticatedRiderId(req: NextRequest): number | null {
  const token = req.cookies.get("f2c_rider_session")?.value;
  const secret = process.env.RIDER_SESSION_SECRET;

  if (!token || !secret || secret.length < 32) return null;

  const parts = token.split(".");
  if (parts.length !== 2) return null;

  try {
    const expected = createHmac("sha256", secret)
      .update(parts[0])
      .digest();

    const supplied = Buffer.from(parts[1], "base64url");

    if (
      supplied.length !== expected.length ||
      !timingSafeEqual(supplied, expected)
    ) {
      return null;
    }

    const payload = JSON.parse(
      Buffer.from(parts[0], "base64url").toString("utf8")
    );

    if (
      !Number.isInteger(payload.riderId) ||
      !Number.isFinite(payload.exp) ||
      payload.exp <= Date.now()
    ) {
      return null;
    }

    return payload.riderId;
  } catch {
    return null;
  }
}

export async function POST(req: NextRequest) {
  try {
    const riderId = getAuthenticatedRiderId(req);

    if (!riderId) {
      return NextResponse.json(
        { success: false, error: "Please log in again" },
        { status: 401 }
      );
    }

    const body = await req.json();
    const assignmentId = Number(body.assignmentId);
    const action = body.action;

    if (!Number.isInteger(assignmentId) || assignmentId <= 0) {
      return NextResponse.json(
        { success: false, error: "Valid assignment ID required" },
        { status: 400 }
      );
    }

    if (!["accept", "reject"].includes(action)) {
      return NextResponse.json(
        { success: false, error: "Invalid action" },
        { status: 400 }
      );
    }

    const rider = await prisma.deliveryBoy.findUnique({
      where: { id: riderId },
      select: { id: true, approvalStatus: true },
    });

    if (!rider || rider.approvalStatus !== "Approved") {
      return NextResponse.json(
        { success: false, error: "Rider is not approved" },
        { status: 403 }
      );
    }

    const result = await prisma.$transaction(async (tx) => {
      const assignment = await tx.deliveryAssignment.findUnique({
        where: { id: assignmentId },
      });

      if (!assignment) {
        return { error: "Assignment not found", status: 404 };
      }

      if (assignment.deliveryBoyId !== riderId) {
        return { error: "This order is not assigned to you", status: 403 };
      }

      if (assignment.status !== "Pending Acceptance") {
        return {
          error: "This assignment has already been processed",
          status: 409,
        };
      }

      const updatedCount = await tx.deliveryAssignment.updateMany({
        where: {
          id: assignmentId,
          deliveryBoyId: riderId,
          status: "Pending Acceptance",
        },
        data:
          action === "accept"
            ? { status: "Accepted", acceptedAt: new Date() }
            : { status: "Rejected" },
      });

      if (updatedCount.count !== 1) {
        return { error: "Assignment changed. Refresh and try again.", status: 409 };
      }

      if (action === "accept") {
        await tx.deliveryBoy.update({
          where: { id: riderId },
          data: { status: "Busy" },
        });
      } else {
        const otherActive = await tx.deliveryAssignment.count({
          where: {
            deliveryBoyId: riderId,
            id: { not: assignmentId },
            status: {
              in: ["Pending Acceptance", "Accepted", "Picked Up", "Out For Delivery"],
            },
          },
        });

        await tx.deliveryBoy.update({
          where: { id: riderId },
          data: { status: otherActive > 0 ? "Busy" : "Available" },
        });
      }

      return {
        success: true,
        message: action === "accept" ? "Order Accepted" : "Order Rejected",
      };
    });

    if ("error" in result) {
      return NextResponse.json(
        { success: false, error: result.error },
        { status: result.status }
      );
    }

    return NextResponse.json(result);
  } catch (error) {
    console.error("DELIVERY ACCEPT ERROR:", error);

    return NextResponse.json(
      { success: false, error: "Failed to process request" },
      { status: 500 }
    );
  }
}

import { NextResponse } from "next/server";
import { randomBytes } from "node:crypto";
import { prisma } from "@/lib/prisma";

export const runtime = "nodejs";

export async function POST(req: Request) {
  try {
    const body = await req.json();
    const mobile = String(body.mobile ?? "").trim();

    if (!/^[0-9+\-\s()]{8,20}$/.test(mobile)) {
      return NextResponse.json({ success: false, error: "Enter a valid registered mobile number." }, { status: 400 });
    }

    const sid = process.env.TWILIO_ACCOUNT_SID;
    const token = process.env.TWILIO_AUTH_TOKEN;
    const from = process.env.TWILIO_FROM_NUMBER;

    if (!sid || !token || !from) {
      return NextResponse.json({
        success: false,
        error: "SMS service is not configured. Set TWILIO_ACCOUNT_SID, TWILIO_AUTH_TOKEN and TWILIO_FROM_NUMBER in .env, then restart the server."
      }, { status: 503 });
    }

    const rider = await prisma.deliveryBoy.findUnique({ where: { mobile } });

    if (!rider) {
      return NextResponse.json({ success: false, error: "No rider account found for this mobile number." }, { status: 404 });
    }

    const temporaryPassword = randomBytes(6).toString("hex");
    const endpoint = `https://api.twilio.com/2010-04-01/Accounts/${sid}/Messages.json`;
    const payload = new URLSearchParams({
      To: mobile,
      From: from,
      Body: `Fresh2Cook Rider: Your temporary password is ${temporaryPassword}. Login and change it immediately. If you did not request this, contact support.`
    });

    const sms = await fetch(endpoint, {
      method: "POST",
      headers: {
        Authorization: `Basic ${Buffer.from(`${sid}:${token}`).toString("base64")}`,
        "Content-Type": "application/x-www-form-urlencoded"
      },
      body: payload.toString(),
      cache: "no-store"
    });

    if (!sms.ok) {
      console.error("RIDER PASSWORD SMS FAILED:", await sms.text());
      return NextResponse.json({ success: false, error: "SMS could not be sent. Check your SMS provider settings and registered sender number." }, { status: 502 });
    }

    await prisma.deliveryBoy.update({
      where: { id: rider.id },
      data: { password: temporaryPassword }
    });

    return NextResponse.json({
      success: true,
      message: "Temporary password sent by SMS. Please log in and change it as soon as possible."
    });
  } catch (error) {
    console.error("RIDER FORGOT PASSWORD ERROR:", error);
    return NextResponse.json({ success: false, error: "Password reset failed. Please try again." }, { status: 500 });
  }
}

        {
          success: false,
          error: "OTP is not active for this delivery",
        },
        { status: 400 }
      );
    }

    if (assignment.otp !== otp) {
      return NextResponse.json(
        { success: false, error: "Invalid OTP" },
        { status: 400 }
      );
    }

    const updated = await prisma.deliveryAssignment.update({
      where: { id: assignmentId },
      data: {
        status: "Delivered",
        deliveredAt: new Date(),
        earnings: 25,
      },
    });

    await prisma.order.update({
      where: { id: assignment.orderId },
      data: {
        status: "Delivered",
        deliveryOtp: null,
      },
    });

    await prisma.deliveryBoy.update({
      where: { id: assignment.deliveryBoyId },
      data: {
        status: "Available",
        completedDeliveries: {
          increment: 1,
        },
        earnings: {
          increment: 25,
        },
      },
    });

    return NextResponse.json({
      success: true,
      delivered: true,
      earning: 25,
      assignment: updated,
      message: "Delivery Completed Successfully",
    });
  } catch (error: any) {
    console.error("RIDER COMPLETE DELIVERY ERROR:", error);

    return NextResponse.json(
      {
        success: false,
        error: error?.message || "Delivery Failed",
      },
      { status: 500 }
    );
  }
}
