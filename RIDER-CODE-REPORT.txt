import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

type Rider = {
  id: number;
  name: string;
};

export async function GET() {
  try {
    const riders: Rider[] =
      await prisma.deliveryBoy.findMany({
        orderBy: {
          name: "asc",
        },
      });

    const data =
      await Promise.all(
        riders.map(
          async (rider: Rider) => {
            const activeOrder =
              await prisma.deliveryAssignment.findFirst(
                {
                  where: {
                    deliveryBoyId:
                      rider.id,
                    status: {
                      in: [
                        "Pending Acceptance",
                        "Accepted",
                        "Picked Up",
                        "Out For Delivery",
                      ],
                    },
                  },
                  orderBy: {
                    assignedAt:
                      "desc",
                  },
                }
              );

            const todayDeliveries =
              await prisma.deliveryAssignment.count(
                {
                  where: {
                    deliveryBoyId:
                      rider.id,
                    status:
                      "Delivered",
                    deliveredAt: {
                      gte: new Date(
                        new Date().setHours(
                          0,
                          0,
                          0,
                          0
                        )
                      ),
                    },
                  },
                }
              );

            return {
              ...rider,

              currentOrder:
                activeOrder?.orderId ||
                null,

              currentStatus:
                activeOrder?.status ||
                "Idle",

              todayDeliveries,
            };
          }
        )
      );

    return NextResponse.json(
      data
    );
  } catch (error) {
    console.error(
      "ADMIN DELIVERY RIDERS ERROR",
      error
    );

    return NextResponse.json(
      {
        error:
          "Failed To Load Riders",
      },
      {
        status: 500,
      }
    );
  }
}

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function GET() {
  try {
    const riders =
      await prisma.deliveryBoy.findMany({
        orderBy: {
          createdAt: "desc",
        },
      });

    const summary = {
      total:
        riders.length,

      approved:
        riders.filter(
          (r: any) =>
            r.approvalStatus ===
            "Approved"
        ).length,

      pending:
        riders.filter(
          (r: any) =>
            r.approvalStatus ===
            "Pending"
        ).length,

      rejected:
        riders.filter(
          (r: any) =>
            r.approvalStatus ===
            "Rejected"
        ).length,
    };

    return NextResponse.json({
      success: true,
      summary,
      riders,
    });
  } catch (error) {
    console.error(
      "RIDER APPROVAL GET ERROR:",
      error
    );

    return NextResponse.json(
      {
        success: false,
        error:
          "Failed To Load Riders",
      },
      {
        status: 500,
      }
    );
  }
}

export async function POST(
  req: Request
) {
  try {
    const body =
      await req.json();

    const riderId =
      Number(
        body.riderId
      );

    const status =
      String(
        body.status || ""
      );

    if (!riderId) {
      return NextResponse.json(
        {
          success: false,
          error:
            "Rider ID Required",
        },
        {
          status: 400,
        }
      );
    }

    if (
      ![
        "Approved",
        "Pending",
        "Rejected",
      ].includes(status)
    ) {
      return NextResponse.json(
        {
          success: false,
          error:
            "Invalid Status",
        },
        {
          status: 400,
        }
      );
    }

    const rider =
      await prisma.deliveryBoy.findUnique(
        {
          where: {
            id: riderId,
          },
        }
      );

    if (!rider) {
      return NextResponse.json(
        {
          success: false,
          error:
            "Rider Not Found",
        },
        {
          status: 404,
        }
      );
    }

    const updatedRider =
      await prisma.deliveryBoy.update(
        {
          where: {
            id: riderId,
          },
          data: {
            approvalStatus:
              status,
          },
        }
      );

    return NextResponse.json({
      success: true,
      message:
        `Rider ${status} Successfully`,
      rider:
        updatedRider,
    });
  } catch (error) {
    console.error(
      "RIDER APPROVAL POST ERROR:",
      error
    );

    return NextResponse.json(
      {
        success: false,
        error:
          "Failed To Update Rider Status",
      },
      {
        status: 500,
      }
    );
  }
}

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(
  req: Request
) {
  try {
    const body = await req.json();

    const assignmentId =
      Number(
        body.assignmentId
      );

    const action =
      body.action;

    if (!assignmentId) {
      return NextResponse.json(
        {
          error:
            "Assignment ID Required",
        },
        {
          status: 400,
        }
      );
    }

    const assignment =
      await prisma.deliveryAssignment.findUnique(
        {
          where: {
            id:
              assignmentId,
          },
        }
      );

    if (!assignment) {
      return NextResponse.json(
        {
          error:
            "Assignment Not Found",
        },
        {
          status: 404,
        }
      );
    }

    if (
      action ===
      "accept"
    ) {
      const updated =
        await prisma.deliveryAssignment.update(
          {
            where: {
              id:
                assignmentId,
            },
            data: {
              status:
                "Accepted",

              acceptedAt:
                new Date(),
            },
          }
        );

      await prisma.deliveryBoy.update(
        {
          where: {
            id:
              assignment.deliveryBoyId,
          },
          data: {
            status:
              "Busy",
          },
        }
      );

      return NextResponse.json({
        success: true,
        message:
          "Order Accepted",
        assignment:
          updated,
      });
    }

    if (
      action ===
      "reject"
    ) {
      const updated =
        await prisma.deliveryAssignment.update(
          {
            where: {
              id:
                assignmentId,
            },
            data: {
              status:
                "Rejected",
            },
          }
        );

      await prisma.deliveryBoy.update(
        {
          where: {
            id:
              assignment.deliveryBoyId,
          },
          data: {
            status:
              "Available",
          },
        }
      );

      return NextResponse.json({
        success: true,
        message:
          "Order Rejected",
        assignment:
          updated,
      });
    }

    return NextResponse.json(
      {
        error:
          "Invalid Action",
      },
      {
        status: 400,
      }
    );
  } catch (error) {
    console.error(
      "DELIVERY ACCEPT ERROR",
      error
    );

    return NextResponse.json(
      {
        error:
          "Failed To Process Request",
      },
      {
        status: 500,
      }
    );
  }
}

              phone: true,
              address: true,
              amount: true,
              status: true,
              createdAt: true,
              deliveryOtp: true,
            },

        : [];

      riderIds.length > 0

              },
            },
            select: {
  id: true,
  name: true,
  mobile: true,
  vehicle: true,
  status: true,
  zoneId: true
},
          })
        : [];

    const orderMap = new Map(
      orders.map((order) => [

                  order.phone,
                address:
                  order.address,

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

function generateOtp() {
  return Math.floor(
    1000 + Math.random() * 9000
  ).toString();
}

export async function POST(
  req: Request
) {
  try {
    const body =
      await req.json();

    const orderId = Number(
      body.orderId
    );

    const riderId = Number(
      body.riderId
    );

    if (
      !orderId ||
      !riderId
    ) {
      return NextResponse.json(
        {
          success: false,
          error:
            "Order and Rider required",
        },
        { status: 400 }
      );
    }

    const otp =
      generateOtp();

    const assignment =
      await prisma.deliveryAssignment.create({
        data: {
          orderId,
          deliveryBoyId:
            riderId,
          otp,
          status:
            "Pending Acceptance",
        },
      });

    await prisma.order.update({
      where: {
        id: orderId,
      },
      data: {
        deliveryOtp:
          otp,
        status:
          "Assigned To Rider",
      },
    });

    return NextResponse.json({
      success: true,
      assignment,
    });
  } catch (error: any) {
    console.error(error);

    return NextResponse.json(
      {
        success: false,
        error:
          error?.message ||
          "Assignment failed",
      },
      { status: 500 }
    );
  }
}
