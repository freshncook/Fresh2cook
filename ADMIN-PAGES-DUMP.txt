"use client";

export const dynamic = "force-dynamic";

import { useEffect, useMemo, useState } from "react";

type Order = {
  id: number;
  customer: string;
  phone?: string | null;
  address?: string | null;
  amount: number;
  status: string;
  createdAt: string;
};

type Feedback = {
  id: number;
  customer: string;
  phone: string;
  rating: number;
  comment: string;
  createdAt: string;
};

type Customer = {
  name: string;
  phone: string;
  address: string;
  orders: number;
  totalSpend: number;
  lastOrder: string;
  status: string;
  rating: number | null;
};

export default function AdminCustomersPage() {
  const [orders, setOrders] = useState<Order[]>([]);
  const [feedback, setFeedback] = useState<Feedback[]>([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("All");
  const [error, setError] = useState("");

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    try {
      setLoading(true);
      setError("");

      const [ordersRes, feedbackRes] = await Promise.all([
        fetch("/api/orders", { cache: "no-store" }),
        fetch("/api/feedback", { cache: "no-store" }),
      ]);

      const ordersData = await ordersRes.json();
      const feedbackData = await feedbackRes.json();

      if (!ordersRes.ok) {
        throw new Error(
          ordersData?.error || "Failed to load orders"
        );
      }

      if (!feedbackRes.ok) {
        throw new Error(
          feedbackData?.error || "Failed to load feedback"
        );
      }

      setOrders(
        Array.isArray(ordersData)
          ? ordersData
          : ordersData?.orders || []
      );

      setFeedback(
        Array.isArray(feedbackData)
          ? feedbackData
          : feedbackData?.feedback || []
      );
    } catch (error) {
      console.error("CUSTOMERS LOAD ERROR:", error);

      setError(
        error instanceof Error
          ? error.message
          : "Failed to load customers"
      );
    } finally {
      setLoading(false);
    }
  }

  const customers = useMemo<Customer[]>(() => {
    const map = new Map<string, Customer>();

    for (const order of orders) {
      const phone = String(order.phone || "").trim();

      const key =
        phone ||
        order.customer.trim().toLowerCase();

      const existing = map.get(key);

      if (existing) {
        existing.orders += 1;
        existing.totalSpend += Number(
          order.amount || 0
        );

        if (
          new Date(order.createdAt).getTime() >
          new Date(existing.lastOrder).getTime()
        ) {
          existing.lastOrder = order.createdAt;
          existing.address =
            order.address || existing.address;
          existing.status = order.status;
        }

        continue;
      }

      const customerReviews = feedback.filter(
        (item) =>
          item.phone === phone ||
          item.customer.trim().toLowerCase() ===
            order.customer.trim().toLowerCase()
      );

      const rating =
        customerReviews.length > 0
          ? customerReviews.reduce(
              (sum, item) =>
                sum + Number(item.rating || 0),
              0
            ) / customerReviews.length
          : null;

      map.set(key, {
        name: order.customer,
        phone,
        address: order.address || "",
        orders: 1,
        totalSpend: Number(order.amount || 0),
        lastOrder: order.createdAt,
        status: order.status,
        rating,
      });
    }

    return Array.from(map.values()).sort(
      (a, b) => b.totalSpend - a.totalSpend
    );
  }, [orders, feedback]);

  const statuses = useMemo(() => {
    const values = customers
      .map((customer) => customer.status)
      .filter(Boolean);

    return [
      "All",
      ...Array.from(new Set(values)),
    ];
  }, [customers]);

  const filteredCustomers = useMemo(() => {
    const query = search.trim().toLowerCase();

    return customers.filter((customer) => {
      const matchesSearch =
        !query ||
        customer.name
          .toLowerCase()
          .includes(query) ||
        customer.phone
          .toLowerCase()
          .includes(query) ||
        customer.address
          .toLowerCase()
          .includes(query);

      const matchesStatus =
        statusFilter === "All" ||
        customer.status === statusFilter;

      return matchesSearch && matchesStatus;
    });
  }, [customers, search, statusFilter]);

  const totalCustomers = customers.length;

  const repeatCustomers = customers.filter(
    (customer) => customer.orders > 1
  ).length;

  const totalCustomerValue = customers.reduce(
    (sum, customer) =>
      sum + customer.totalSpend,
    0
  );

  const averageOrderValue =
    orders.length > 0
      ? Math.round(
          orders.reduce(
            (sum, order) =>
              sum + Number(order.amount || 0),
            0
          ) / orders.length
        )
      : 0;

  const activeCustomers = customers.filter(
    (customer) =>
      !["Delivered", "Cancelled"].includes(
        customer.status
      )
  ).length;

  function openWhatsApp(customer: Customer) {
    const cleanedPhone =
      customer.phone.replace(/\D/g, "");

    if (!cleanedPhone) {
      alert(
        "Customer mobile number is not available."
      );
      return;
    }

    const number =
      cleanedPhone.length === 10
        ? `91${cleanedPhone}`
        : cleanedPhone;

      `Hello ${customer.name},`,
      "",
      "Thank you for being a Fresh2Cook customer! ??",
      "",
      "How can we help you today?",
    ].join("\n");

    window.open(
      `https://wa.me/${number}?text=${encodeURIComponent(
        message
      )}`,
      "_blank",
      "noopener,noreferrer"
    );
  }

  function formatDate(date: string) {
    if (!date) return "�";

    const parsed = new Date(date);

    if (Number.isNaN(parsed.getTime())) {
      return "�";
    }

    return parsed.toLocaleDateString(
      "en-IN",
      {
        day: "2-digit",
        month: "short",
        year: "numeric",
      }
    );
  }

  function getStatusClass(status: string) {
    const value = status.toLowerCase();

    if (value.includes("deliver")) {
      return "bg-green-50 text-green-700 border-green-200";
    }

    if (value.includes("cancel")) {
      return "bg-red-50 text-red-700 border-red-200";
    }

    if (value.includes("assign")) {
      return "bg-blue-50 text-blue-700 border-blue-200";
    }

    if (value.includes("process")) {
      return "bg-purple-50 text-purple-700 border-purple-200";
    }

    return "bg-orange-50 text-orange-700 border-orange-200";
  }

  return (
    <div className="min-h-screen bg-[#f4f7f7] p-4 md:p-6">
      <div className="max-w-[1500px] mx-auto space-y-6">

        {/* HEADER */}
        <div className="bg-white border border-gray-200 rounded-2xl shadow-sm overflow-hidden">

          <div className="bg-gradient-to-r from-[#063f3d] to-[#0b625c] px-6 py-7 text-white">

            <div className="flex flex-col lg:flex-row lg:items-center lg:justify-between gap-5">

              <div className="flex items-center gap-4">
                <div className="w-12 h-12 rounded-xl bg-white/10 border border-white/20 flex items-center justify-center text-2xl">
                  ??
                </div>

                <div>
                  <h1 className="text-2xl md:text-3xl font-bold">
                    Customer Management
                  </h1>

                  <p className="text-sm text-white/70 mt-1">
                    Customers, orders, spending and activity
                  </p>
                </div>
              </div>

              <button
                onClick={loadData}
                className="bg-white text-[#063f3d] px-5 py-3 rounded-xl font-bold text-sm hover:bg-gray-100 transition"
              >
                ? Refresh Customers
              </button>

            </div>
          </div>

          {/* KPI */}
          <div className="grid grid-cols-2 lg:grid-cols-5 divide-x divide-gray-100">

            <div className="p-5">
              <p className="text-xs font-bold uppercase tracking-wider text-gray-400">
                Total Customers
              </p>
              <p className="text-2xl font-bold text-[#063f3d] mt-1">
                {totalCustomers}
              </p>
            </div>

            <div className="p-5">
              <p className="text-xs font-bold uppercase tracking-wider text-gray-400">
                Repeat Customers
              </p>
              <p className="text-2xl font-bold text-[#159447] mt-1">
                {repeatCustomers}
              </p>
            </div>

            <div className="p-5">
              <p className="text-xs font-bold uppercase tracking-wider text-gray-400">
                Customer Value
              </p>
              <p className="text-2xl font-bold text-blue-600 mt-1">
                ?{totalCustomerValue.toLocaleString("en-IN")}
              </p>
            </div>

            <div className="p-5">
              <p className="text-xs font-bold uppercase tracking-wider text-gray-400">
                Avg Order Value
              </p>
              <p className="text-2xl font-bold text-purple-600 mt-1">
                ?{averageOrderValue.toLocaleString("en-IN")}
              </p>
            </div>

            <div className="p-5">
              <p className="text-xs font-bold uppercase tracking-wider text-gray-400">
                Active Customers
              </p>
              <p className="text-2xl font-bold text-orange-600 mt-1">
                {activeCustomers}
              </p>
            </div>

          </div>
        </div>

        {/* ERROR */}

    useState<any[]>([]);

  const [name, setName] =
    useState("");

  const [stock, setStock] =
    useState("");

  const [unit, setUnit] =
    useState("Kg");

  useEffect(() => {
    loadInventory();
  }, []);

  function loadInventory() {
    const data = JSON.parse(
      localStorage.getItem(
        "freshcut-inventory"
      ) || "[]"
    );

    setItems(data);
  }

  function addItem() {
    if (!name || !stock) {
      alert(
        "Fill All Fields"
      );
      return;
    }

    const item = {
      id: Date.now(),
      name,
      stock,
      unit,
      updatedAt:
        new Date().toLocaleDateString(),
    };

    const updated = [
      item,
      ...items,
    ];

    localStorage.setItem(
      "freshcut-inventory",
      JSON.stringify(updated)
    );

    setItems(updated);

    setName("");
    setStock("");

    alert(
      "Inventory Added"
    );
  }

  function updateStock(
    id: number,
    value: string
  ) {
    const updated =
      items.map((item) =>
        item.id === id
          ? {
              ...item,
              stock: value,
              updatedAt:
                new Date().toLocaleDateString(),
            }
          : item
      );

    localStorage.setItem(
      "freshcut-inventory",
      JSON.stringify(updated)
    );

    setItems(updated);
  }

  function deleteItem(
    id: number
  ) {
    if (
      !confirm(
        "Delete Item?"
      )
    )
      return;

    const updated =
      items.filter(
        (item) =>
          item.id !== id
      );

    localStorage.setItem(
      "freshcut-inventory",
      JSON.stringify(updated)
    );

    setItems(updated);
  }

  return (

              value={name}

                setName(
                  e.target.value
                )
              }
              placeholder="Item Name"
              className="border p-3 rounded-xl"
            />

            <input
              value={stock}
              onChange={(e) =>
                setStock(
                  e.target.value
                )
              }
              placeholder="Stock"
              className="border p-3 rounded-xl"
            />

            <select
              value={unit}
              onChange={(e) =>
                setUnit(
                  e.target.value
                )
              }
              className="border p-3 rounded-xl"
            >
              <option>Kg</option>
              <option>Gram</option>

"use client";

export const dynamic = "force-dynamic";

import { useEffect, useState } from "react";

type Settings = {
  businessName: string;
  phone: string;
  email: string;
  address: string;
  deliveryCharge: string;
  minOrder: string;
};

export default function SettingsPage() {
  const [settings, setSettings] = useState<Settings>({
    businessName: "Fresh2Cook",
    phone: "9811001901",
    email: "info@freshcut.com",
    address: "New Delhi",
    deliveryCharge: "40",
    minOrder: "199",
  });

  useEffect(() => {
    try {
      const data = localStorage.getItem("freshcut-settings");
      if (data) setSettings(JSON.parse(data));
    } catch {
      // keep defaults
    }
  }, []);

  function update(key: keyof Settings, value: string) {
    setSettings((current) => ({
      ...current,
      [key]: value,
    }));
  }

  function saveSettings() {
    localStorage.setItem("freshcut-settings", JSON.stringify(settings));
    alert("Settings Saved Successfully");
  }

  return (
    <div className="min-h-screen bg-[#f5f7f7] p-5 md:p-7">
      <div className="mx-auto max-w-[1200px]">
        <div className="mb-7">
          <p className="text-xs font-bold uppercase tracking-[0.18em] text-emerald-700">
            Fresh2Cook Admin
          </p>
          <h1 className="mt-1 text-3xl font-extrabold text-[#123b38] md:text-4xl">
            System Settings
          </h1>
          <p className="mt-1 text-sm text-slate-500">
            Manage core business and delivery configuration.
          </p>
        </div>

        <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm md:p-8">
          <div className="mb-7 border-b border-slate-100 pb-5">
            <h2 className="text-lg font-extrabold text-[#123b38]">
              Business Configuration
            </h2>
            <p className="mt-1 text-sm text-slate-500">
              Update the information used across the system.
            </p>
          </div>

          <div className="grid gap-5 md:grid-cols-2">
            <Field
              label="Business Name"
              value={settings.businessName}
              onChange={(value) => update("businessName", value)}
            />

            <Field
              label="Mobile Number"
              value={settings.phone}
              onChange={(value) => update("phone", value)}
            />

            <Field
              label="Email Address"
              value={settings.email}
              onChange={(value) => update("email", value)}
            />

            <Field
              label="Delivery Charge"
              value={settings.deliveryCharge}
              onChange={(value) => update("deliveryCharge", value)}
            />

            <Field
              label="Minimum Order Value"
              value={settings.minOrder}
              onChange={(value) => update("minOrder", value)}
            />

            <div className="md:col-span-2">
              <label className="mb-2 block text-sm font-bold text-slate-600">
                Business Address
              </label>
              <textarea
                value={settings.address}
                onChange={(e) => update("address", e.target.value)}
                rows={4}
                className="w-full rounded-xl border border-slate-300 px-4 py-3 text-sm outline-none focus:border-emerald-600"
              />
            </div>
          </div>

            className="mt-7 rounded-xl bg-[#063f3d] px-10 py-3 text-sm font-bold text-white hover:bg-[#0a514d]"
          >
            Save Settings
          </button>
        </div>
      </div>
    </div>
  );
}

function Field({
  label,
  value,
  onChange,
}: {
  label: string;
  value: string;
  onChange: (value: string) => void;
}) {
  return (
    <div>
      <label className="mb-2 block text-sm font-bold text-slate-600">
        {label}
      </label>
      <input
        value={value}
        onChange={(e) => onChange(e.target.value)}
        className="w-full rounded-xl border border-slate-300 px-4 py-3 text-sm outline-none focus:border-emerald-600"
      />
