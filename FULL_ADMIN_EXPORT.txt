"use client";

import { useEffect, useMemo, useState } from "react";

export default function OrdersPage() {
  const [orders, setOrders] = useState<any[]>([]);
  const [boys, setBoys] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");
  const [statusFilter, setStatusFilter] = useState("All");

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    try {
      const [ordersRes, boysRes] = await Promise.all([
        fetch("/api/orders"),
        fetch("/api/delivery-boys"),
      ]);

      const ordersData = await ordersRes.json();
      const boysData = await boysRes.json();

      setOrders(Array.isArray(ordersData) ? ordersData : ordersData?.orders || []);
      setBoys(
        Array.isArray(boysData)
          ? boysData
          : boysData?.riders || []
      );
    } catch (error) {
      console.error(error);
    } finally {
      setLoading(false);
    }
  }

  async function assignOrder(orderId: number, riderId: number) {
    if (!riderId) return;

    try {
      const res = await fetch("/api/delivery-control", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          orderId,
          riderId,
        }),
      });

      const data = await res.json();

      if (!res.ok) {
        alert(data.error || "Assignment Failed");
        return;
      }

      alert("Order Assigned Successfully");
      loadData();
    } catch (error) {
      console.error(error);
      alert("Something Went Wrong");
    }
  }

  const statuses = useMemo(() => {
    const values = orders
      .map((order) => order.status)
      .filter(Boolean);

    return ["All", ...Array.from(new Set(values))];
  }, [orders]);

  const filteredOrders = useMemo(() => {
    const keyword = search.toLowerCase().trim();

    return orders.filter((order) => {
      const matchesSearch =
        !keyword ||
        String(order.id || "").toLowerCase().includes(keyword) ||
        String(order.customer || "").toLowerCase().includes(keyword) ||
        String(order.phone || "").toLowerCase().includes(keyword) ||
        String(order.address || "").toLowerCase().includes(keyword);

      const matchesStatus =
        statusFilter === "All" ||
        order.status === statusFilter;

      return matchesSearch && matchesStatus;
    });
  }, [orders, search, statusFilter]);

  const totalAmount = orders.reduce(
    (sum, order) => sum + Number(order.amount || 0),
    0
  );

  const pendingCount = orders.filter(
    (order) =>
      !["Delivered", "Cancelled"].includes(order.status)
  ).length;

  const assignedCount = orders.filter((order) =>
    String(order.status || "")
      .toLowerCase()
      .includes("assigned")
  ).length;

  if (loading) {
    return (
      <div className="min-h-[70vh] flex items-center justify-center bg-[#f4f7f7]">
        <div className="bg-white rounded-2xl shadow-sm border border-gray-200 px-8 py-6 text-center">
          <div className="w-10 h-10 border-4 border-[#159447] border-t-transparent rounded-full animate-spin mx-auto" />
          <p className="mt-4 text-sm font-semibold text-gray-600">
            Loading Orders...
          </p>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-[#f4f7f7] p-4 md:p-6">
      <div className="max-w-[1500px] mx-auto space-y-6">

        {/* HEADER */}
        <div className="bg-white border border-gray-200 rounded-2xl shadow-sm overflow-hidden">
          <div className="bg-gradient-to-r from-[#063f3d] to-[#0b625c] px-6 py-7 text-white">
            <div className="flex flex-col lg:flex-row lg:items-center lg:justify-between gap-5">
              <div>
                <div className="flex items-center gap-3">
                  <div className="w-11 h-11 rounded-xl bg-white/10 border border-white/20 flex items-center justify-center text-2xl">
                    ??
                  </div>

                  <div>
                    <h1 className="text-2xl md:text-3xl font-bold">
                      Orders Management
                    </h1>

                    <p className="text-sm text-white/70 mt-1">
                      Manage orders and assign delivery riders
                    </p>
                  </div>
                </div>
              </div>

              <button
                onClick={loadData}
                className="bg-white text-[#063f3d] px-5 py-3 rounded-xl font-bold text-sm hover:bg-gray-100 transition"
              >
                ? Refresh Orders
              </button>
            </div>
          </div>

          {/* STATS */}
          <div className="grid grid-cols-2 lg:grid-cols-4 divide-x divide-gray-100">
            <div className="p-5">
              <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
                Total Orders
              </p>
              <p className="text-2xl font-bold text-[#063f3d] mt-1">
                {orders.length}
              </p>
            </div>

            <div className="p-5">
              <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
                Active Orders

                {assignedCount}

        {/* ORDERS */}
        <div className="space-y-4">

          {filteredOrders.map((order: any) => {

"use client";

import { useEffect, useState } from "react";

export default function DeliveryPage() {
  const [riders, setRiders] = useState<any[]>([]);
  const [alerts, setAlerts] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    loadData();

    const interval = setInterval(() => {
      loadData();
    }, 10000);

    return () => clearInterval(interval);
  }, []);

  async function loadData() {
    try {
      const riderRes = await fetch(
        "/api/delivery-live-status"
      );

      const riderData =
        await riderRes.json();

      setRiders(
        Array.isArray(riderData)
          ? riderData
          : []
      );

      const alertRes = await fetch(
        "/api/admin-delivery-alerts"
      );

      const alertData =
        await alertRes.json();

      const pendingAlerts =
        alertData.alerts || [];

      setAlerts(pendingAlerts);

function timer(
  deadline: string | null
) {
  if (!deadline)
    return "30:00";

  const seconds =
    Math.max(
      0,
      Math.ceil(
        (
          new Date(
            deadline
          ).getTime() -
          Date.now()
        ) / 1000
      )
    );

  const m =
    Math.floor(
      seconds / 60
    );

  const s =
    seconds % 60;

  return `${String(
    m
  ).padStart(
    2,
    "0"
  )}:${String(
    s
  ).padStart(
    2,
    "0"
  )}`;

            <div className="text-2xl font-black text-slate-900">
              {readyCount}
            </div>
          </div>

        </div>


        {error && (
          <div className="mb-4 rounded-xl border border-red-200 bg-red-50 px-4 py-3 text-sm font-bold text-red-700">
            {error}
          </div>
        )}


        {/* QUEUE */}

        <div className="space-y-3">

          {loading ? (
            <div className="rounded-xl bg-white p-8 text-center text-sm text-slate-500">
              Loading processing queue...
            </div>
          ) : (
            orders.map(
              (order) => {

                const accepted =
                  !!order.processingAcceptedById;

                const mine =
                  order.processingAcceptedById ===
                  staff?.id;

                const allPacked =
                  order.items.length >
                    0 &&
                  order.items.every(
                    (item) =>
                      item.packed
                  );

                const prepared =
                  !!order.processingPreparedAt;

                const cross =
                  !!order.processingCrossCheckedAt;

                const sticker =
                  !!order.processingStickerAt;

                const packaging =
                  !!order.processingPackagingAt;

                const step4Ready =
                  prepared &&
                  cross &&
                  sticker &&
                  packaging;

                const isProcessing =
                  order.status ===
                  "Processing";

                const isReady =
                  order.status ===
                  "Ready For Dispatch";

                return (

                  <div
                    key={
                      order.id
                    }
                    className="overflow-hidden rounded-xl border border-slate-200 bg-white shadow-sm"
                  >

                    {/* ORDER TOP ROW */}

                    <div className="grid items-center gap-3 border-b border-slate-100 bg-slate-50 px-3 py-3 md:grid-cols-[90px_1fr_140px_150px_130px]">

                      <div>
                        <div className="text-[10px] font-bold uppercase text-slate-400">
                          Order
                        </div>

                        <div className="text-base font-black text-slate-900">
                          #{order.id}
                        </div>
                      </div>


                      <div>
                        <div className="text-sm font-black text-slate-900">
                          {order.customer}
                        </div>

                        <div className="text-[11px] text-slate-500">
                          {order.phone ||
                            "-"}
                        </div>
                      </div>


                      <div>
                        <div className="text-[10px] font-bold uppercase text-slate-400">
                          Value
                        </div>

                        <div className="font-black text-slate-800">
                          {money(
                            order.amount
                          )}
                        </div>
                      </div>


                      <div>
                        {!accepted ? (
                          <label className="flex cursor-pointer items-center gap-2 text-sm font-black text-emerald-700">
                            <input
                              type="checkbox"
                              checked={false}
                              onChange={() =>
                                action(
                                  order.id,
                                  "accept"
                                )
                              }
                              disabled={
                                busyOrder ===
                                order.id
                              }
                              className="h-5 w-5 accent-emerald-600"
                            />

                            ACCEPT
                          </label>
                        ) : (
                          <div>
                            <div className="text-[10px] font-bold uppercase text-slate-400">
                              Accepted By
                            </div>

                            <div className="text-sm font-black text-emerald-700">
                              {
                                order.acceptedByName
                              }
                            </div>
                          </div>
                        )}
                      </div>


                      <div className="text-right">

                        {isProcessing ? (
                          <div>
                            <div className="text-[10px] font-bold uppercase text-slate-400">
                              SLA
                            </div>

                            <div
                              className={`font-mono text-lg font-black ${
                                new Date(
                                  order.processingDeadline ||
                                    0
                                ).getTime() <
                                tick
                                  ? "text-red-600"
                                  : "text-orange-600"
                              }`}
                            >
                              {timer(
                                order.processingDeadline
                              )}
                            </div>
                          </div>
                        ) : isReady ? (
                          <span className="rounded-full bg-blue-100 px-3 py-1 text-[11px] font-black text-blue-700">
                            READY
                          </span>
                        ) : (
                          <span className="rounded-full bg-slate-100 px-3 py-1 text-[11px] font-bold text-slate-600">
                            {order.status}
                          </span>
                        )}

                      </div>

                    </div>


                    {/* STEPS */}

                    <div className="grid grid-cols-2 border-b border-slate-100 md:grid-cols-4">

                      {/* STEP 1 */}

                      <div
                        className={`border-r border-b p-3 md:border-b-0 ${
                          accepted
                            ? "bg-emerald-50"
                            : "bg-white"
                        }`}
                      >

                        <div className="text-[10px] font-black uppercase text-slate-400">
                          Step 1
                        </div>

                        <div className="mt-1 flex items-center gap-2">

                          <span
                            className={`flex h-6 w-6 items-center justify-center rounded-full text-xs font-black ${
                              accepted
                                ? "bg-emerald-600 text-white"
                                : "bg-slate-200 text-slate-500"
                            }`}
                          >
                            {accepted
                              ? "?"
                              : "1"}
                          </span>

                          <div>
                            <div className="text-xs font-black">
                              Accept
                            </div>

                            {order.acceptedByName && (
                              <div className="text-[10px] text-slate-500">
                                {
                                  order.acceptedByName
                                }{" "}
                                ï{" "}
                                {timeText(
                                  order.processingAcceptedAt
                                )}
                              </div>
                            )}
                          </div>

                        </div>

                      </div>


                      {/* STEP 2 */}

                      <div
                        className={`border-r border-b p-3 md:border-b-0 ${
                          prepared
                            ? "bg-emerald-50"
                            : "bg-white"
                        }`}
                      >

                        <div className="text-[10px] font-black uppercase text-slate-400">
                          Step 2
                        </div>

                        <label className="mt-1 flex items-center gap-2">

                          <input
                            type="checkbox"
                            checked={
                              prepared
                            }
                            disabled={
                              !mine ||
                              !allPacked ||
                              !!prepared
                            }
                            onChange={() =>
                              action(
                                order.id,
                                "prepare"
                              )
                            }
                            className="h-5 w-5 accent-emerald-600"
                          />

                          <span className="text-xs font-black">
                            Prepare & Pack
                          </span>

                        </label>

                      </div>


                      {/* STEP 3 */}

                      <div
                        className={`border-r border-b p-3 md:border-b-0 ${
                          step4Ready
                            ? "bg-emerald-50"
                            : "bg-white"
                        }`}
                      >

                        <div className="text-[10px] font-black uppercase text-slate-400">
                          Step 3
                        </div>

                        <div className="mt-1 space-y-1">

                          <label className="flex items-center gap-2 text-[11px] font-bold">
                            <input
                              type="checkbox"
                              checked={
                                !!checks[
                                  order.id
                                ]?.cross ||
                                cross
                              }
                              disabled={
                                !mine ||
                                !prepared ||
                                cross
                              }
                              onChange={(e) =>
                                setChecks(
                                  (
                                    current
                                  ) => ({
                                    ...current,
                                    [order.id]:
                                      {
                                        cross:
                                          e
                                            .target
                                            .checked,
                                        sticker:
                                          !!current[
                                            order.id
                                          ]?.sticker,
                                        packaging:
                                          !!current[
                                            order.id
                                          ]?.packaging,
                                      },
                                  })
                                )
                              }
                              className="h-4 w-4 accent-emerald-600"
                            />
                            Qty / Product Check
                          </label>

                          <label className="flex items-center gap-2 text-[11px] font-bold">
                            <input
                              type="checkbox"
                              checked={
                                !!checks[
                                  order.id
                                ]?.sticker ||
                                sticker
                              }
                              disabled={
                                !mine ||
                                !prepared ||
                                sticker
                              }
                              onChange={(e) =>
                                setChecks(
                                  (
                                    current
                                  ) => ({
                                    ...current,
                                    [order.id]:
                                      {
                                        cross:
                                          !!current[
                                            order.id
                                          ]?.cross,
                                        sticker:
                                          e
                                            .target
                                            .checked,
                                        packaging:
                                          !!current[
                                            order.id
                                          ]?.packaging,
                                      },
                                  })
                                )
                              }
                              className="h-4 w-4 accent-emerald-600"
                            />
                            Sticker / Label
                          </label>

                          <label className="flex items-center gap-2 text-[11px] font-bold">
                            <input
                              type="checkbox"
                              checked={
                                !!checks[
                                  order.id
                                ]?.packaging ||
                                packaging
                              }
                              disabled={
                                !mine ||
                                !prepared ||
                                packaging
                              }
                              onChange={(e) =>
                                setChecks(
                                  (
                                    current
                                  ) => ({
                                    ...current,
                                    [order.id]:
                                      {
                                        cross:
                                          !!current[
                                            order.id
                                          ]?.cross,
                                        sticker:
                                          !!current[
                                            order.id
                                          ]?.sticker,
                                        packaging:
                                          e
                                            .target
                                            .checked,
                                      },
                                  })
                                )
                              }
                              className="h-4 w-4 accent-emerald-600"
                            />
                            Packaging
                          </label>

                          {!cross &&
                            !sticker &&
                            !packaging &&
                            prepared &&
                            (() => {
                              const c =
                                checks[
                                  order.id
                                ];

                              if (
                                c?.cross &&
                                c?.sticker &&
                                c?.packaging
                              ) {
                                action(
                                  order.id,
                                  "crossCheck",
                                  {
                                    crossChecked:
                                      true,
                                    stickerChecked:
                                      true,
                                    packagingChecked:
                                      true,
                                  }
                                );
                              }

                              return null;
                            })()}

                        </div>

                      </div>


                      {/* STEP 4 */}

                      <div
                        className={`p-3 ${
                          isReady
                            ? "bg-blue-50"
                            : "bg-white"
                        }`}
                      >

                        <div className="text-[10px] font-black uppercase text-slate-400">
                          Step 4
                        </div>

                        {isReady ? (

                          <div className="mt-2 text-xs font-black text-blue-700">
                            ? SENT FOR DISPATCH
                          </div>

                        ) : (

                          <button
                            type="button"
                            disabled={
                              !mine ||
                              !step4Ready ||
                              busyOrder ===
                                order.id
                            }
                            onClick={() =>
                              action(
                                order.id,
                                "dispatch"
                              )
                            }
                            className={`mt-1 w-full rounded-lg px-3 py-2 text-xs font-black text-white ${
                              mine &&
                              step4Ready
                                ? "bg-blue-600 hover:bg-blue-700"
                                : "cursor-not-allowed bg-slate-300"
                            }`}
                          >
                            ?? Send For Dispatch
                          </button>

                        )}

                      </div>

                    </div>


                    {/* PRODUCT GRID */}

                    <div className="border-t border-slate-100">

                      <button
                        type="button"
                        onClick={() =>
                          setExpanded(
                            expanded ===
                              order.id
                              ? null
                              : order.id
                          )
                        }
                        className="flex w-full items-center justify-between px-3 py-2 text-left hover:bg-slate-50"
                      >

                        <div>
                          <span className="text-xs font-black text-slate-700">
                            Packing List
                          </span>

                          <span className="ml-2 text-[10px] text-slate-400">
                            {order.items.length} products
                          </span>
                        </div>

                        <span className="text-xs font-bold text-emerald-600">
                          {expanded ===
                          order.id
                            ? "Hide"
                            : "View"}
                        </span>

                      </button>


                      {expanded ===
                        order.id && (

                        <div className="overflow-x-auto px-3 pb-3">

                          <table className="w-full min-w-[700px] border-collapse text-xs">

                            <thead>

                              <tr className="bg-slate-900 text-white">

                                <th className="w-12 border border-slate-700 px-2 py-2 text-center">
                                  SR
                                </th>

                                <th className="border border-slate-700 px-3 py-2 text-left">
                                  PRODUCT
                                </th>

                                <th className="w-36 border border-slate-700 px-3 py-2 text-left">
                                  PACK / WEIGHT
                                </th>

                                <th className="w-16 border border-slate-700 px-2 py-2 text-center">
                                  QTY
                                </th>

                                <th className="w-28 border border-slate-700 px-3 py-2 text-right">
                                  AMOUNT
                                </th>

                                <th className="w-28 border border-slate-700 px-2 py-2 text-center">
                                  PACKED
                                </th>

                              </tr>

                            </thead>


                            <tbody>

                              {order.items.map(
                                (
                                  item,
                                  index
                                ) => (

                                  <tr
                                    key={
                                      item.id
                                    }
                                    className={
                                      item.packed
                                        ? "bg-emerald-50"
                                        : "bg-white"
                                    }
                                  >

                                    <td className="border border-slate-200 px-2 py-3 text-center font-bold text-slate-400">
                                      {
                                        index +
                                        1
                                      }
                                    </td>

                                    <td className="border border-slate-200 px-3 py-3 font-black text-slate-800">
                                      {
                                        item.productName
                                      }
                                    </td>

                                    <td className="border border-slate-200 px-3 py-3 font-semibold text-slate-600">
                                      {
                                        item.pack
                                      }
                                    </td>

                                    <td className="border border-slate-200 px-2 py-3 text-center font-black">
                                      {
                                        item.quantity
                                      }
                                    </td>

                                    <td className="border border-slate-200 px-3 py-3 text-right font-bold">
                                      {money(
                                        item.total
                                      )}
                                    </td>

                                    <td className="border border-slate-200 px-2 py-3 text-center">

                                      <input
                                        type="checkbox"
                                        checked={
                                          item.packed
                                        }
                                        disabled={
                                          !mine ||
                                          busyItem ===
                                            item.id ||
                                          prepared
                                        }
                                        onChange={(
                                          e
                                        ) =>
                                          packItem(
                                            order.id,
                                            item.id,
                                            e
                                              .target
                                              .checked
                                          )
                                        }
                                        className="h-5 w-5 accent-emerald-600"
                                      />

                                    </td>

                                  </tr>

                                )
                              )}

                            </tbody>

                          </table>

                        </div>

                      )}

                    </div>

                  </div>

                );
              }
            )
          )}

        </div>

      </div>

    </main>
  );
}
