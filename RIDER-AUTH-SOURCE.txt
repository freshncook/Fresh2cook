========== app/api/rider-register/route.ts ==========

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(
  req: Request
) {
  try {
    const body =
      await req.json();

    const name =
      String(
        body.name || ""
      ).trim();

    const mobile =
      String(
        body.mobile || ""
      ).trim();

    const password =
      String(
        body.password || ""
      ).trim();

    const vehicle =
      String(
        body.vehicle || ""
      ).trim();

    if (
      !name ||
      !mobile ||
      !vehicle
    ) {
      return NextResponse.json(
        {
          success: false,
          error:
            "Name, Mobile And Vehicle Required",
        },
        {
          status: 400,
        }
      );
    }

    const existingRider =
      await prisma.deliveryBoy.findUnique(
        {
          where: {
            mobile,
          },
        }
      );

    if (
      existingRider
    ) {
      return NextResponse.json(
        {
          success: false,
          error:
            "Mobile Number Already Registered",
        },
        {
          status: 400,
        }
      );
    }

    const rider =
      await prisma.deliveryBoy.create(
        {
          data: {
            name,
            mobile,
            password,
            vehicle,

            address:
              body.address ||
              null,

            aadhaar:
              body.aadhaar ||
              null,

            photo:
              body.photo ||
              null,

            aadhaarFront:
              body.aadhaarFront ||
              null,

            aadhaarBack:
              body.aadhaarBack ||
              null,

            aadharFile:
              body.aadharFile ||
              null,

            licenseFile:
              body.licenseFile ||
              null,

            rcFile:
              body.rcFile ||
              null,

            approvalStatus:
              "Pending",

            status:
              "Offline",

            rating: 5,

            earnings: 0,

            completedDeliveries: 0,
          },
        }
      );

    return NextResponse.json({
      success: true,
      message:
        "Registration Submitted Successfully",

      rider: {
        id:
          rider.id,

        name:
          rider.name,

        mobile:
          rider.mobile,

        vehicle:
          rider.vehicle,

        approvalStatus:
          rider.approvalStatus,
      },
    });
  } catch (error) {
    console.error(
      "RIDER REGISTER ERROR:",
      error
    );

    return NextResponse.json(
      {
        success: false,
        error:
          "Registration Failed",
      },
      {
        status: 500,
      }
    );
  }
}






========== app/api/delivery-accept/route.ts ==========

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(
  req: Request
) {
  try {
    const body = await req.json();

    const assignmentId =
      Number(
        body.assignmentId
      );

    const action =
      body.action;

    if (!assignmentId) {
      return NextResponse.json(
        {
          error:
            "Assignment ID Required",
        },
        {
          status: 400,
        }
      );
    }

    const assignment =
      await prisma.deliveryAssignment.findUnique(
        {
          where: {
            id:
              assignmentId,
          },
        }
      );

    if (!assignment) {
      return NextResponse.json(
        {
          error:
            "Assignment Not Found",
        },
        {
          status: 404,
        }
      );
    }

    if (
      action ===
      "accept"
    ) {
      const updated =
        await prisma.deliveryAssignment.update(
          {
            where: {
              id:
                assignmentId,
            },
            data: {
              status:
                "Accepted",

              acceptedAt:
                new Date(),
            },
          }
        );

      await prisma.deliveryBoy.update(
        {
          where: {
            id:
              assignment.deliveryBoyId,
          },
          data: {
            status:
              "Busy",
          },
        }
      );

      return NextResponse.json({
        success: true,
        message:
          "Order Accepted",
        assignment:
          updated,
      });
    }

    if (
      action ===
      "reject"
    ) {
      const updated =
        await prisma.deliveryAssignment.update(
          {
            where: {
              id:
                assignmentId,
            },
            data: {
              status:
                "Rejected",
            },
          }
        );

      await prisma.deliveryBoy.update(
        {
          where: {
            id:
              assignment.deliveryBoyId,
          },
          data: {
            status:
              "Available",
          },
        }
      );

      return NextResponse.json({
        success: true,
        message:
          "Order Rejected",
        assignment:
          updated,
      });
    }

    return NextResponse.json(
      {
        error:
          "Invalid Action",
      },
      {
        status: 400,
      }
    );
  } catch (error) {
    console.error(
      "DELIVERY ACCEPT ERROR",
      error
    );

    return NextResponse.json(
      {
        error:
          "Failed To Process Request",
      },
      {
        status: 500,
      }
    );
  }
}

}







========== app/api/rider-orders/route.ts ==========

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function GET(req: Request) {
  try {
    const { searchParams } =
      new URL(req.url);

    const riderId = Number(
      searchParams.get("riderId")
    );

    if (!riderId) {
      return NextResponse.json(
        {
          success: false,
          error: "Rider ID required",
        },
        {
          status: 400,
        }
      );
    }

    const assignments =
      await prisma.deliveryAssignment.findMany({
        where: {
          deliveryBoyId: riderId,
        },
        orderBy: {
          assignedAt: "desc",
        },
      });

    const orderIds =
      assignments.map(
        (a) => a.orderId
      );

    const orders =
      orderIds.length > 0
        ? await prisma.order.findMany({
            where: {
              id: {
                in: orderIds,
              },
            },
          })
        : [];

    const orderMap = new Map(
      orders.map((o) => [o.id, o])
    );

    const result =
      assignments.map((a) => ({
        assignmentId: a.id,
        orderId: a.orderId,
        otp: a.otp,
        status: a.status,
        earnings: a.earnings,
        assignedAt: a.assignedAt,

        customer:
          orderMap.get(a.orderId)
            ?.customer || "",

        phone:
          orderMap.get(a.orderId)
            ?.phone || "",

        address:
          orderMap.get(a.orderId)
            ?.address || "",

        amount:
          orderMap.get(a.orderId)
            ?.amount || 0,
      }));

    return NextResponse.json({
      success: true,
      orders: result,
    });
  } catch (error: any) {
    console.error(
      "RIDER ORDERS ERROR",
      error
    );

    return NextResponse.json(
      {
        success: false,
        error: error?.message ?? "Internal Server Error",
      },
      {
        status: 500,
      }
    );
  }
}
