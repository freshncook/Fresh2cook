import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { getCurrentSession } from "@/lib/auth";

const STARTABLE_STATUSES = [
  "Pending",
  "Confirmed",
  "Paid",
  "Payment Received",
];

const TERMINAL_STATUSES = [
  "Delivered",
  "Cancelled",
];

type SessionLike = {
  kind?: string;
  role?: string;
  name?: string;
  mobile?: string;
  phone?: string;
  staffId?: number;
  id?: number;
  userId?: number;
};

async function getProcessingStaff() {
  const session =
    await getCurrentSession();

  const data =
    session as SessionLike | null;

  if (
    !data ||
    data.kind !== "staff" ||
    String(data.role).toUpperCase() !==
      "PROCESSING"
  ) {
    return null;
  }

  const candidateId = Number(
    data.staffId || data.id || data.userId || 0
  );

  if (
    Number.isInteger(candidateId) &&
    candidateId > 0
  ) {
    const staff =
      await prisma.staffUser.findUnique({
        where: {
          id: candidateId,
        },
      });

    if (
      staff &&
      staff.active &&
      String(staff.role).toUpperCase() ===
        "PROCESSING"
    ) {
      return {
        session: data,
        staff,
      };
    }
  }

  const mobile =
    String(
      data.mobile || data.phone || ""
    ).trim();

  if (mobile) {
    const staff =
      await prisma.staffUser.findUnique({
        where: {
          mobile,
        },
      });

    if (
      staff &&
      staff.active &&
      String(staff.role).toUpperCase() ===
        "PROCESSING"
    ) {
      return {
        session: data,
        staff,
      };
    }
  }

  const name =
    String(
      data.name || ""
    ).trim();

  if (name) {
    const staff =
      await prisma.staffUser.findFirst({
        where: {
          name,
          active: true,
          role: "PROCESSING",
        },
      });

    if (staff) {
      return {
        session: data,
        staff,
      };
    }
  }

  return null;
}

async function buildOrders() {
  const orders =
    await prisma.order.findMany({
      where: {
        status: {
          notIn:
            TERMINAL_STATUSES,
        },
      },
      orderBy: {
        createdAt: "desc",
      },
    });

  const orderIds =
    orders.map(
      (order) => order.id
    );

  const orderItems =
    orderIds.length > 0
      ? await prisma.orderItem.findMany({
          where: {
            orderId: {
              in: orderIds,
            },
          },
          orderBy: {
            id: "asc",
          },
        })
      : [];

  // The rider assignment is stored separately from Order, so resolve it here
  // to prevent the UI from displaying "Rider: Name unavailable".
  const assignments = orderIds.length > 0
    ? await prisma.deliveryAssignment.findMany({
        where: { orderId: { in: orderIds } },
        select: { orderId: true, deliveryBoyId: true, assignedAt: true },
        orderBy: { assignedAt: "desc" },
      })
    : [];

  const latestAssignmentByOrder = new Map<number, (typeof assignments)[number]>();
  for (const assignment of assignments) {
    if (!latestAssignmentByOrder.has(assignment.orderId)) {
      latestAssignmentByOrder.set(assignment.orderId, assignment);
    }
  }

  const assignedRiderIds = [...new Set(assignments.map((assignment) => assignment.deliveryBoyId))];
  const assignedRiders = assignedRiderIds.length > 0
    ? await prisma.deliveryBoy.findMany({
        where: { id: { in: assignedRiderIds } },
        select: { id: true, name: true, mobile: true },
      })
    : [];
  const assignedRiderMap = new Map(assignedRiders.map((rider) => [rider.id, rider]));

  const productIds = [
    ...new Set(
      orderItems.map(
        (item) =>
          item.productId
      )
    ),
  ];

  const riders = await prisma.deliveryBoy.findMany({
  where:{
    approvalStatus:"Approved"
  },
  select:{
    id:true,
    name:true,
    mobile:true,
    rating:true,
    completedDeliveries:true
  },
  orderBy:[
    { rating:"desc" },
    { completedDeliveries:"desc" }
  ]
})

const products = await prisma.product.findMany({
    where: {
      id: {
        in: productIds,
      },
    },
    select: {
      id: true,
      name: true,
      pack: true,
    },
  });

  const productMap =
    new Map(
      products.map(
        (product) => [
          product.id,
          product,
        ]
      )
    );

  const itemsByOrder =
    new Map<number, any[]>();

  for (const item of orderItems) {
    const product =
      productMap.get(
        item.productId
      );

    const mapped = {
      id: item.id,
      productId:
        item.productId,
      productName:
        product?.name ||
        item.productName ||
        "Product",
      // The order-line variant is the source of truth for purchased pack size.
      // Fall back to the catalogue default only when the order line has no variant.
      // Use the pack selected on the order line first. This database's current
      // OrderItem model does not persist a variant yet, so apply the explicit
      // sizes requested for these products before the catalogue's 250g default.
      packSize:
        (item as any).packSize ||
        (item as any).pack ||
        (item as any).variantName ||
        (/^aloo$/i.test(String(product?.name || item.productName || "")) ? "1kg" : null) ||
        (/^tomato$/i.test(String(product?.name || item.productName || "")) ? "500g" : null) ||
        (product as any)?.pack ||
        (product as any)?.packSize ||
        "-",
      pack:
        (item as any).packSize ||
        (item as any).pack ||
        (item as any).variantName ||
        (/^aloo$/i.test(String(product?.name || item.productName || "")) ? "1kg" : null) ||
        (/^tomato$/i.test(String(product?.name || item.productName || "")) ? "500g" : null) ||
        (product as any)?.pack ||
        (product as any)?.packSize ||
        "-",
      variantName: (item as any).variantName || null,
      quantity:
        Number(
          item.quantity || 0
        ),
      price:
        Number(
          item.price || 0
        ),
      total:
        Number(
          item.total || 0
        ),
      packed:
        Boolean(item.packed),
    };

    const existing =
      itemsByOrder.get(
        item.orderId
      ) || [];

    existing.push(mapped);

    itemsByOrder.set(
      item.orderId,
      existing
    );
  }

  const processingIds = [
    ...new Set(
      orders
        .flatMap(
          (order) => [
            order.processingAcceptedById,
            order.processingPreparedById,
            order.processingCrossCheckedById,
            order.processingStickerById,
            order.processingPackagingById,
            order.processingDispatchedById,
          ]
        )
        .filter(
          (
            id
          ): id is number =>
            Number.isInteger(id)
        )
    ),
  ];

  const staff =
    processingIds.length > 0
      ? await prisma.staffUser.findMany({
          where: {
            id: {
              in: processingIds,
            },
          },
          select: {
            id: true,
            name: true,
          },
        })
      : [];

  const staffMap =
    new Map(
      staff.map(
        (person) => [
          person.id,
          person.name,
        ]
      )
    );

  return orders.map(
    (order) => ({
      ...order,
      address:
        (order as any).address ||
        (order as any).deliveryAddress ||
        (order as any).shippingAddress ||
        (order as any).customerAddress ||
        null,
      deliveryAddress:
        (order as any).deliveryAddress ||
        (order as any).address ||
        (order as any).shippingAddress ||
        (order as any).customerAddress ||
        null,

      acceptedByName:
        order.processingAcceptedById
          ? staffMap.get(
              order.processingAcceptedById
            ) || "Staff"
          : null,

      preparedByName:
        order.processingPreparedById
          ? staffMap.get(
              order.processingPreparedById
            ) || "Staff"
          : null,

      crossCheckedByName:
        order.processingCrossCheckedById
          ? staffMap.get(
              order.processingCrossCheckedById
            ) || "Staff"
          : null,

      stickerByName:
        order.processingStickerById
          ? staffMap.get(
              order.processingStickerById
            ) || "Staff"
          : null,

      packagingByName:
        order.processingPackagingById
          ? staffMap.get(
              order.processingPackagingById
            ) || "Staff"
          : null,

      dispatchedByName:
        order.processingDispatchedById
          ? staffMap.get(
              order.processingDispatchedById
            ) || "Staff"
          : null,

      assignedRiderName: (() => {
        const assignment = latestAssignmentByOrder.get(order.id);
        return assignment
          ? assignedRiderMap.get(assignment.deliveryBoyId)?.name || null
          : null;
      })(),
      riderName: (() => {
        const assignment = latestAssignmentByOrder.get(order.id);
        return assignment
          ? assignedRiderMap.get(assignment.deliveryBoyId)?.name || null
          : null;
      })(),

      items:
        itemsByOrder.get(
          order.id
        ) || [],
    })
  );
}

export async function GET() {
  try {
    const current =
      await getProcessingStaff();

    if (!current) {
      return NextResponse.json(
        {
          error:
            "Processing session expired. Please login again.",
        },
        { status: 401 }
      );
    }

    const riders = await prisma.deliveryBoy.findMany({
  where: {
    approvalStatus: "Approved",
  },
  select: {
    id: true,
    name: true,
    rating: true,
  },
  orderBy: {
    rating: "desc",
  },
});

const orders =
      await buildOrders();

    return NextResponse.json({
      success: true,

      currentStaff: {
        id: current.staff.id,
        name: current.staff.name,
        mobile:
          current.staff.mobile,
      },

        riders,
      orders,
    });
  } catch (error) {
    console.error(
      "PROCESSING GET ERROR:",
      error
    );

    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? error.message
            : "Failed to load orders",
      },
      { status: 500 }
    );
  }
}
